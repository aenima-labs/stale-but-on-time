\newcommand{\tlbox}[5][]{% [style] x0 x1 y label
  \draw[draw=hair, line width=0.4pt, rounded corners=1.2pt, #1]
    (#2,#4-0.26) rectangle (#3,#4+0.26);
  \node[font=\scriptsize, text=ink] at ({(#2+#3)/2},#4) {#5};
}
